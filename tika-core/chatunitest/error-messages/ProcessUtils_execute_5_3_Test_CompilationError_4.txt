package org.apache.tika.utils;

import org.apache.tika.utils.FileProcessResult;
import org.apache.tika.utils.ProcessUtils;
import org.apache.tika.utils.StreamGobbler;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.concurrent.TimeoutException;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class ProcessUtils_execute_5_3_Test {

    @Mock
    private StreamGobbler errGobbler;

    @InjectMocks
    private ProcessUtils processUtils;

    private Path stdoutRedirect;

    private ProcessBuilder pb;

    private long timeoutMillis;

    private int maxStdErrBuffer;

    @BeforeEach
    public void setUp() throws IOException {
        stdoutRedirect = Files.createTempFile("test", ".txt");
        pb = new ProcessBuilder("echo", "test");
        timeoutMillis = 1000;
        maxStdErrBuffer = 1024;
    }

    @Test
    public void testExecute() throws IOException, InterruptedException, TimeoutException {
        when(errGobbler.getStreamLength()).thenReturn(4L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(java.util.Arrays.asList("test"));
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(4, result.stderrLength);
        assertEquals(4, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("test", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecuteWithTimeout() throws IOException, InterruptedException, TimeoutException {
        when(errGobbler.getStreamLength()).thenReturn(4L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(java.util.Arrays.asList("test"));
        Process mockProcess = mock(Process.class);
        when(mockProcess.waitFor(1000, TimeUnit.MILLISECONDS)).thenReturn(false);
        when(mockProcess.exitValue()).thenReturn(1);
        when(mockProcess.getErrorStream()).thenReturn(new ByteArrayInputStream("test".getBytes()));
        when(mockProcess.getInputStream()).thenReturn(new ByteArrayInputStream("test".getBytes()));
        when(processUtils.register(mockProcess)).thenReturn("mockId");
        when(processUtils.release("mockId")).thenReturn(mockProcess);
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(4, result.stderrLength);
        assertEquals(4, result.stdoutLength);
        assertTrue(result.isTimeout);
        assertEquals(-1, result.exitValue);
        assertEquals("test", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecuteWithException() throws IOException, InterruptedException, TimeoutException {
        when(errGobbler.getStreamLength()).thenReturn(4L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(java.util.Arrays.asList("test"));
        Process mockProcess = mock(Process.class);
        when(mockProcess.waitFor(1000, TimeUnit.MILLISECONDS)).thenThrow(new InterruptedException());
        when(mockProcess.exitValue()).thenReturn(1);
        when(mockProcess.getErrorStream()).thenReturn(new ByteArrayInputStream("test".getBytes()));
        when(mockProcess.getInputStream()).thenReturn(new ByteArrayInputStream("test".getBytes()));
        when(processUtils.register(mockProcess)).thenReturn("mockId");
        when(processUtils.release("mockId")).thenReturn(mockProcess);
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(4, result.stderrLength);
        assertEquals(4, result.stdoutLength);
        assertTrue(result.isTimeout);
        assertEquals(-1000, result.exitValue);
        assertEquals("test", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecutePrivateMethods() throws Exception {
        Process mockProcess = mock(Process.class);
        String mockId = UUID.randomUUID().toString();
        when(processUtils.register(mockProcess)).thenReturn(mockId);
        when(processUtils.release(mockId)).thenReturn(mockProcess);
        ReflectionTestUtils.invokeMethod(processUtils, "register", mockProcess);
    }
}
