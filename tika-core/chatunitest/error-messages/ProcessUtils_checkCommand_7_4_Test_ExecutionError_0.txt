    @BeforeEach
    public void setUp() {
        processUtils = new ProcessUtils();

    @Test
    public void testCheckCommandWithNoErrorValue() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        when(Runtime.getRuntime().exec(checkCmd)).thenReturn(mockProcess);
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenReturn(true);
        when(mockProcess.exitValue()).thenReturn(0);
        boolean result = processUtils.checkCommand(checkCmd);
        assertTrue(result);
        verify(mockProcess, times(1)).destroyForcibly();
    }

    @Test
    public void testCheckCommandWithErrorValue() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        int[] errorValue = { 1 };
        when(Runtime.getRuntime().exec(checkCmd)).thenReturn(mockProcess);
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenReturn(true);
        when(mockProcess.exitValue()).thenReturn(1);
        boolean result = processUtils.checkCommand(checkCmd, errorValue);
        assertFalse(result);
        verify(mockProcess, times(1)).destroyForcibly();
    }

    @Test
    public void testCheckCommandWithTimeout() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        when(Runtime.getRuntime().exec(checkCmd)).thenReturn(mockProcess);
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenReturn(false);
        boolean result = processUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).destroyForcibly();
    }

    @Test
    public void testCheckCommandWithIOException() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        when(Runtime.getRuntime().exec(checkCmd)).thenThrow(new IOException("IO Exception"));
        boolean result = processUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).destroyForcibly();
    }

    @Test
    public void testCheckCommandWithInterruptedException() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        when(Runtime.getRuntime().exec(checkCmd)).thenReturn(mockProcess);
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenThrow(new InterruptedException("Interrupted Exception"));
        boolean result = processUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).destroyForcibly();
    }

    @Test
    public void testCheckCommandWithTimeoutException() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        when(Runtime.getRuntime().exec(checkCmd)).thenReturn(mockProcess);
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenThrow(new TimeoutException("Timeout Exception"));
        boolean result = processUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).destroyForcibly();
    }

    @Test
    public void testCheckCommandWithSecurityException() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        when(Runtime.getRuntime().exec(checkCmd)).thenThrow(new SecurityException("Security Exception"));
        assertThrows(SecurityException.class, () -> processUtils.checkCommand(checkCmd));
    }

    @Test
    public void testCheckCommandWithErrorInErrorValue() throws IOException, InterruptedException, TimeoutException {
        String[] checkCmd = { "ls", "-l" };
        int[] errorValue = { 127 };
        when(Runtime.getRuntime().exec(checkCmd)).thenReturn(mockProcess);
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenReturn(true);
        when(mockProcess.exitValue()).thenReturn(127);
        boolean result = processUtils.checkCommand(checkCmd, errorValue);
        assertFalse(result);
        verify(mockProcess, times(1)).destroyForcibly();
    }

    public void testCheckCommandWithNonStandardError() throws IOException, InterruptedException, TimeoutException {

        when(Runtime.getRuntime().exec(checkCmd)).thenReturn(mockProcess);
