@ExtendWith(MockitoExtension.class)
public class ProcessUtils_execute_4_1_Test {

    @Mock
    private StreamGobbler outGobbler;

    @Mock
    private StreamGobbler errGobbler;

    @InjectMocks
    private ProcessUtils processUtils;

    @BeforeEach
    public void setUp() {
        MockitoAnnotations.openMocks(this);
    }

    @Test
    public void testExecute() throws IOException, InterruptedException {
        // Arrange
        ProcessBuilder pb = mock(ProcessBuilder.class);
        when(pb.start()).thenReturn(mock(Process.class));
        long timeoutMillis = 1000;
        int maxStdoutBuffer = 1024;
        int maxStdErrBuffer = 1024;
        when(outGobbler.getStreamLength()).thenReturn(100L);
        when(outGobbler.getIsTruncated()).thenReturn(false);
        when(outGobbler.getLines()).thenReturn(List.of("line1", "line2"));
        when(errGobbler.getStreamLength()).thenReturn(50L);
        when(errGobbler.getIsTruncated()).thenReturn(true);
        when(errGobbler.getLines()).thenReturn(List.of("error1"));
        Thread outThread = mock(Thread.class);
        Thread errThread = mock(Thread.class);
        when(outThread.join(1000)).thenReturn(null);
        when(errThread.join(1000)).thenReturn(null);
        when(outThread.interrupt()).thenReturn(null);
        when(errThread.interrupt()).thenReturn(null);
        // Act
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, maxStdoutBuffer, maxStdErrBuffer);
        // Assert
        assertNotNull(result);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(100L, result.stderrLength);
        assertEquals(50L, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("line1\nline2", result.stdout);
        assertEquals("error1", result.stderr);
        assertTrue(result.stdoutTruncated);
        assertTrue(result.stderrTruncated);
    }

    @Test

        when(outGobbler.getLines()).thenReturn(List.of("line1", "line2"));
        when(errGobbler.getStreamLength()).thenReturn(50L);
        when(errGobbler.getIsTruncated()).thenReturn(true);
        when(errGobbler.getLines()).thenReturn(List.of("error1"));
        Thread outThread = mock(Thread.class);
        Thread errThread = mock(Thread.class);
        when(outThread.join(1000)).thenReturn(null);
        when(errThread.join(1000)).thenReturn(null);
        when(outThread.interrupt()).thenReturn(null);
        when(errThread.interrupt()).thenReturn(null);
        when(pb.waitFor(timeoutMillis, TimeUnit.MILLISECONDS)).thenReturn(false);
        // Act
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, maxStdoutBuffer, maxStdErrBuffer);
        // Assert
        assertNotNull(result);
        assertEquals(100, result.processTimeMillis);
        assertEquals(100L, result.stderrLength);
        assertEquals(50L, result.stdoutLength);
        assertTrue(result.isTimeout);
        assertEquals(-1, result.exitValue);
        assertEquals("line1\nline2", result.stdout);
        assertEquals("error1", result.stderr);
        assertTrue(result.stdoutTruncated);
        assertTrue(result.stderrTruncated);
    }
