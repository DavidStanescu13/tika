package org.apache.tika.utils;

import org.apache.tika.utils.FileProcessResult;
import org.apache.tika.utils.ProcessUtils;
import org.apache.tika.utils.StreamGobbler;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.util.concurrent.ConcurrentHashMap;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.UUID;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class ProcessUtils_execute_4_2_Test {

    @InjectMocks
    private ProcessUtils processUtils;

    @Mock
    private ProcessBuilder processBuilder;

    @Mock
    private Process process;

    @Mock
    private StreamGobbler outGobbler;

    @Mock
    private StreamGobbler errGobbler;

    @BeforeEach
    public void setUp() {
        when(processBuilder.start()).thenReturn(process);
        when(process.getInputStream()).thenReturn(mock(InputStream.class));
        when(process.getErrorStream()).thenReturn(mock(InputStream.class));
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(true);
        when(process.exitValue()).thenReturn(0);
    }

    @Test
    public void testExecute_Success() throws IOException, InterruptedException {
        when(outGobbler.getLines()).thenReturn(List.of("line1", "line2"));
        when(outGobbler.getStreamLength()).thenReturn(10L);
        when(outGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(List.of("error1", "error2"));
        when(errGobbler.getStreamLength()).thenReturn(20L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        FileProcessResult result = processUtils.execute(processBuilder, 1000, 100, 100);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(20, result.stderrLength);
        assertEquals(10, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("line1\nline2", result.stdout);
        assertEquals("error1\nerror2", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecute_Timeout() throws IOException, InterruptedException {
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(false);
        FileProcessResult result = processUtils.execute(processBuilder, 1000, 100, 100);
        assertTrue(result.isTimeout);
    }

    @Test
    public void testExecute_InterruptedException() throws IOException, InterruptedException {
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenThrow(new InterruptedException());
        FileProcessResult result = processUtils.execute(processBuilder, 1000, 100, 100);
        assertEquals(-1000, result.exitValue);
    }
}
