@ExtendWith(MockitoExtension.class)
public class ProcessUtils_checkCommand_6_4_Test {

    @Mock
    private Process mockProcess;

    @BeforeEach
    public void setUp() throws Exception {
        ConcurrentHashMap<String, Process> processMap = new ConcurrentHashMap<>();
        Field field = ProcessUtils.class.getDeclaredField("PROCESS_MAP");
        field.setAccessible(true);
        field.set(null, processMap);
    }

    @Test
    public void testCheckCommandWithError() throws Exception {
        when(mockProcess.waitFor()).thenReturn(1);
        when(mockProcess.exitValue()).thenReturn(1);
        when(mockProcess.getInputStream()).thenReturn(new ByteArrayInputStream("error".getBytes()));
        when(mockProcess.getErrorStream()).thenReturn(new ByteArrayInputStream("error".getBytes()));
        boolean result = ProcessUtils.checkCommand("checkCmd", 1);
        assertFalse(result);
    }

    @Test
    public void testCheckCommandWithNoErrorValue() throws Exception {
        when(mockProcess.waitFor()).thenReturn(0);
        when(mockProcess.exitValue()).thenReturn(0);
        when(mockProcess.getInputStream()).thenReturn(new ByteArrayInputStream("output".getBytes()));
        when(mockProcess.getErrorStream()).thenReturn(new ByteArrayInputStream("".getBytes()));
        boolean result = ProcessUtils.checkCommand("checkCmd");
        assertTrue(result);
    }

    @Test
    public void testCheckCommandWithTimeoutException() throws Exception {
        when(mockProcess.waitFor()).thenThrow(new TimeoutException());
        boolean result = ProcessUtils.checkCommand("checkCmd", 0);
        assertFalse(result);
    }

    @Test
    public void testCheckCommandWithIOException() throws Exception {
        when(mockProcess.waitFor()).thenThrow(new IOException());
        boolean result = ProcessUtils.checkCommand("checkCmd", 0);
        assertFalse(result);
    }

    @Test
    public void testCheckCommandWithSecurityException() throws Exception {
        when(mockProcess.waitFor()).thenThrow(new SecurityException());
        boolean result = ProcessUtils.checkCommand("checkCmd", 0);
        assertFalse(result);
    }

    @Test
    public void testCheckCommandWithErrorValue() throws Exception {
        when(mockProcess.waitFor()).thenReturn(1);
        when(mockProcess.exitValue()).thenReturn(1);
        when(mockProcess.getInputStream()).thenReturn(new ByteArrayInputStream("error".getBytes()));
        when(mockProcess.getErrorStream()).thenReturn(new ByteArrayInputStream("error".getBytes()));
        boolean result = ProcessUtils.checkCommand("checkCmd", 1);
        assertFalse(result);
    }

    @Test
    public void testCheckCommandWithInterruptedException() throws Exception {
        when(mockProcess.waitFor()).thenThrow(new InterruptedException());
        boolean result = ProcessUtils.checkCommand("checkCmd", 0);
        assertFalse(result);
    }
}
