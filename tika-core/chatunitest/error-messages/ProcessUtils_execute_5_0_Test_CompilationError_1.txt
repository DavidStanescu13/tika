@ExtendWith(MockitoExtension.class)
public class ProcessUtils_execute_5_0_Test {

    @Mock
    private ProcessBuilder processBuilder;

    @Mock
    private Process process;

    @Mock
    private StreamGobbler errGobbler;

    @InjectMocks
    private ProcessUtils processUtils;

    @BeforeEach
    public void setUp() {
        when(processBuilder.start()).thenReturn(process);
        when(process.getErrorStream()).thenReturn(mock(InputStream.class));
        when(process.getInputStream()).thenReturn(mock(InputStream.class));
        when(process.getOutputStream()).thenReturn(mock(InputStream.class));
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(true);
        when(process.exitValue()).thenReturn(0);
    }

    @Test
    public void testExecute() throws IOException {
        Path stdoutRedirect = Paths.get("test.txt");
        long timeoutMillis = 1000;
        int maxStdErrBuffer = 1024;
        FileProcessResult result = processUtils.execute(processBuilder, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertNotNull(result);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(0, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("", result.stdout);
        assertEquals("", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecuteWithTimeout() throws IOException, InterruptedException {
        Path stdoutRedirect = Paths.get("test.txt");
        long timeoutMillis = 1;
        int maxStdErrBuffer = 1024;
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(false);
        FileProcessResult result = processUtils.execute(processBuilder, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertNotNull(result);
        assertTrue(result.isTimeout);
        assertEquals(1, result.processTimeMillis);
        assertEquals(0, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertEquals(-1, result.exitValue);
        assertEquals("", result.stdout);
        assertEquals("", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecuteWithException() throws IOException, InterruptedException {
        Path stdoutRedirect = Paths.get("test.txt");
        long timeoutMillis = 1000;
        int maxStdErrBuffer = 1024;
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenThrow(new InterruptedException());
        FileProcessResult result = processUtils.execute(processBuilder, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertNotNull(result);
        assertTrue(result.isTimeout);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(0, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertEquals(-1000, result.exitValue);
        assertEquals("", result.stdout);
        assertEquals("", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }
}
