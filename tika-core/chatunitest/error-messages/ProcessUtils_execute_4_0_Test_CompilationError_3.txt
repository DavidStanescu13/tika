    @BeforeEach
    public void setUp() {

        when(process.getErrorStream()).thenReturn(mock(InputStream.class));

        when(process.exitValue()).thenReturn(0);
    }

    @Test
    public void testExecute() throws IOException, InterruptedException {
        ProcessUtils processUtils = new ProcessUtils();
        long timeoutMillis = 1000;
        int maxStdoutBuffer = 1024;
        int maxStdErrBuffer = 1024;
        FileProcessResult result = processUtils.execute(processBuilder, timeoutMillis, maxStdoutBuffer, maxStdErrBuffer);
        assertNotNull(result);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(0, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertTrue(result.stdout.isEmpty());
        assertTrue(result.stderr.isEmpty());
    }

    @Test
    public void testExecuteWithTimeout() throws IOException, InterruptedException {
        ProcessUtils processUtils = new ProcessUtils();
        long timeoutMillis = 100;
        int maxStdoutBuffer = 1024;
        int maxStdErrBuffer = 1024;
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(false);
        FileProcessResult result = processUtils.execute(processBuilder, timeoutMillis, maxStdoutBuffer, maxStdErrBuffer);
        assertNotNull(result);
        assertTrue(result.processTimeMillis < 100);
        assertEquals(0, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertTrue(result.isTimeout);
        assertEquals(-1, result.exitValue);
        assertTrue(result.stdout.isEmpty());
        assertTrue(result.stderr.isEmpty());
    }

    @Test
    public void testExecuteWithException() throws IOException, InterruptedException {
        ProcessUtils processUtils = new ProcessUtils();
        long timeoutMillis = 1000;
        int maxStdoutBuffer = 1024;
        int maxStdErrBuffer = 1024;
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenThrow(new InterruptedException());

        assertNotNull(result);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(0, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(-1000, result.exitValue);
        assertTrue(result.stdout.isEmpty());
        assertTrue(result.stderr.isEmpty());
    }
