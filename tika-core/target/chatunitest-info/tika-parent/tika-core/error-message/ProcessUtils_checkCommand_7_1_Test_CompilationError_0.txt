@ExtendWith(MockitoExtension.class)
public class ProcessUtils_checkCommand_7_1_Test {

    @Mock
    private Process mockProcess;

    @Mock
    private StreamGobbler mockOutGobbler;

    @Mock
    private StreamGobbler mockErrGobbler;

    @BeforeEach
    public void setUp() {
        when(mockProcess.getInputStream()).thenReturn(mockOutGobbler);
        when(mockProcess.getErrorStream()).thenReturn(mockErrGobbler);
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenReturn(true);
        when(mockProcess.exitValue()).thenReturn(0);
    }

    @Test
    public void testCheckCommandWithNoErrorValue() throws Exception {
        String[] checkCmd = { "ls", "-l" };
        boolean result = ProcessUtils.checkCommand(checkCmd);
        assertTrue(result);
        verify(mockProcess, times(1)).waitFor(60000, TimeUnit.MILLISECONDS);
        verify(mockProcess, times(1)).exitValue();
    }

    @Test
    public void testCheckCommandWithErrorValue() throws Exception {
        String[] checkCmd = { "ls", "-l" };
        int[] errorValue = { 1 };
        boolean result = ProcessUtils.checkCommand(checkCmd, errorValue);
        assertTrue(result);
        verify(mockProcess, times(1)).waitFor(60000, TimeUnit.MILLISECONDS);
        verify(mockProcess, times(1)).exitValue();
    }

    @Test
    public void testCheckCommandWithTimeoutException() throws Exception {
        String[] checkCmd = { "ls", "-l" };
        when(mockProcess.waitFor(60000, TimeUnit.MILLISECONDS)).thenReturn(false);
        boolean result = ProcessUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).waitFor(60000, TimeUnit.MILLISECONDS);
        verify(mockProcess, times(1)).exitValue();
    }

    @Test
    public void testCheckCommandWithIOException() throws Exception {
        String[] checkCmd = { "ls", "-l" };
        doThrow(new IOException()).when(mockProcess).waitFor(60000, TimeUnit.MILLISECONDS);
        boolean result = ProcessUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).waitFor(60000, TimeUnit.MILLISECONDS);
        verify(mockProcess, times(1)).exitValue();
    }

    @Test
    public void testCheckCommandWithInterruptedException() throws Exception {
        String[] checkCmd = { "ls", "-l" };
        doThrow(new InterruptedException()).when(mockProcess).waitFor(60000, TimeUnit.MILLISECONDS);
        boolean result = ProcessUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).waitFor(60000, TimeUnit.MILLISECONDS);
        verify(mockProcess, times(1)).exitValue();
    }

    @Test
    public void testCheckCommandWithSecurityException() throws Exception {
        String[] checkCmd = { "ls", "-l" };
        doThrow(new SecurityException()).when(mockProcess).waitFor(60000, TimeUnit.MILLISECONDS);
        assertThrows(SecurityException.class, () -> ProcessUtils.checkCommand(checkCmd));
        verify(mockProcess, times(1)).waitFor(60000, TimeUnit.MILLISECONDS);
        verify(mockProcess, times(1)).exitValue();
    }

    @Test
    public void testCheckCommandWithError() throws Exception {
        String[] checkCmd = { "ls", "-l" };
        when(mockProcess.exitValue()).thenReturn(1);
        boolean result = ProcessUtils.checkCommand(checkCmd);
        assertFalse(result);
        verify(mockProcess, times(1)).waitFor(60000, TimeUnit.MILLISECONDS);
        verify(mockProcess, times(1)).exitValue();
    }
}
