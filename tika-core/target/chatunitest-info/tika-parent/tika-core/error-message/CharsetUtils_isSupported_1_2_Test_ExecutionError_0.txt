public class CharsetUtils_isSupported_1_2_Test {

    private CharsetUtils charsetUtils;

    private Method getCharsetICU;

    private Method isSupportedICU;

    @BeforeEach
    public void setUp() throws Exception {
        charsetUtils = new CharsetUtils();
        getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        getCharsetICU.setAccessible(true);
        isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        isSupportedICU.setAccessible(true);
    }

    @Test
    public void testIsSupportedWithICUSupported() throws Exception {
        when(isSupportedICU.invoke(null, "UTF-8")).thenReturn(true);
        assertTrue(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithICUUnsupported() throws Exception {
        when(isSupportedICU.invoke(null, "UTF-8")).thenReturn(false);
        assertTrue(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithICUException() throws Exception {
        when(isSupportedICU.invoke(null, "UTF-8")).thenThrow(new InvocationTargetException(new Exception()));
        assertTrue(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithCharsetSupported() throws Exception {
        when(Charset.isSupported("UTF-8")).thenReturn(true);
        assertTrue(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithCharsetUnsupported() throws Exception {
        when(Charset.isSupported("UTF-8")).thenReturn(false);
        assertFalse(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithIllegalCharsetNameException() throws Exception {
        when(Charset.isSupported("UTF-8")).thenThrow(new IllegalCharsetNameException("UTF-8"));
        assertFalse(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithIllegalArgumentException() throws Exception {
        when(Charset.isSupported("UTF-8")).thenThrow(new IllegalArgumentException());
        assertFalse(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithUnexpectedException() throws Exception {
        when(Charset.isSupported("UTF-8")).thenThrow(new Exception());
        assertFalse(charsetUtils.isSupported("UTF-8"));
    }
}
