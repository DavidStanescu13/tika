package org.apache.tika.utils;

import org.apache.tika.utils.FileProcessResult;
import org.apache.tika.utils.ProcessUtils;
import org.apache.tika.utils.StreamGobbler;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.concurrent.ConcurrentHashMap;

import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;

import java.util.UUID;

@ExtendWith(MockitoExtension.class)
public class ProcessUtils_execute_5_3_Test {

    @Mock
    private ProcessBuilder processBuilder;

    @Mock
    private Process process;

    @Mock
    private StreamGobbler errGobbler;

    @InjectMocks
    private ProcessUtils processUtils;

    @BeforeEach
    public void setUp() {
        when(processBuilder.start()).thenReturn(process);
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(true);
        when(process.exitValue()).thenReturn(0);
        when(process.getErrorStream()).thenReturn(mock(InputStream.class));
        when(process.getInputStream()).thenReturn(mock(InputStream.class));
        when(process.getOutputStream()).thenReturn(mock(InputStream.class));
        when(process.destroyForcibly()).thenReturn(null);
        when(process.isAlive()).thenReturn(false);
        when(errGobbler.getStreamLength()).thenReturn(100L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(List.of("line1", "line2"));
    }

    @Test
    public void testExecute() throws IOException, InterruptedException {
        Path stdoutRedirect = Paths.get("test.txt");
        Files.createFile(stdoutRedirect);
        FileProcessResult result = processUtils.execute(processBuilder, 1000, stdoutRedirect, 1000);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(100, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("line1\nline2", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
        verify(processBuilder, times(1)).start();
        verify(process, times(1)).waitFor(1000, TimeUnit.MILLISECONDS);
        verify(process, times(1)).exitValue();
        verify(process, times(1)).destroyForcibly();
        verify(errGobbler, times(1)).getStreamLength();
        verify(errGobbler, times(1)).getIsTruncated();
        verify(errGobbler, times(1)).getLines();
    }

    @Test
    public void testExecuteWithTimeout() throws IOException, InterruptedException {
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(false);
        Path stdoutRedirect = Paths.get("test.txt");
        Files.createFile(stdoutRedirect);
        FileProcessResult result = processUtils.execute(processBuilder, 1000, stdoutRedirect, 1000);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(100, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertTrue(result.isTimeout);
        assertEquals(-1, result.exitValue);
        assertEquals("", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
        verify(processBuilder, times(1)).start();
        verify(process, times(1)).waitFor(1000, TimeUnit.MILLISECONDS);
        verify(process, times(1)).destroyForcibly();
        verify(errGobbler, times(1)).getStreamLength();
        verify(errGobbler, times(1)).getIsTruncated();
        verify(errGobbler, times(1)).getLines();
    }

    @Test
    public void testExecuteWithException() throws IOException, InterruptedException {
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenThrow(new InterruptedException());
        Path stdoutRedirect = Paths.get("test.txt");
        Files.createFile(stdoutRedirect);
        FileProcessResult result = processUtils.execute(processBuilder, 1000, stdoutRedirect, 1000);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(100, result.stderrLength);
        assertEquals(0, result.stdoutLength);
        assertTrue(result.isTimeout);
        assertEquals(-1000, result.exitValue);
        assertEquals("", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }
}
