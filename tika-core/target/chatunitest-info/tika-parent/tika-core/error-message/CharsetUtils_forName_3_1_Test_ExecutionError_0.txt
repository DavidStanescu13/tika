public class CharsetUtils_forName_3_1_Test {

    private CharsetUtils charsetUtils;

    private Map<String, Charset> commonCharsets;

    private Method getCharsetICU;

    private Method isSupportedICU;

    @BeforeEach
    public void setUp() throws Exception {
        charsetUtils = new CharsetUtils();
        commonCharsets = new HashMap<>();
        commonCharsets.put("utf-8", Charset.forName("UTF-8"));
        commonCharsets.put("iso-8859-1", Charset.forName("ISO-8859-1"));
        commonCharsets.put("cp1252", Charset.forName("CP1252"));
        commonCharsets.put("windows-1252", Charset.forName("Windows-1252"));
        getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        getCharsetICU.setAccessible(true);
        isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        isSupportedICU.setAccessible(true);
    }

    @Test
    public void testForNameWithNull() {
        assertThrows(IllegalArgumentException.class, () -> {
            charsetUtils.forName(null);
        });
    }

    @Test
    public void testForNameWithInvalidName() {
        assertThrows(IllegalCharsetNameException.class, () -> {
            charsetUtils.forName("invalid");
        });
    }

    @Test
    public void testForNameWithValidName() {
        assertEquals(Charset.forName("UTF-8"), charsetUtils.forName("utf-8"));
        assertEquals(Charset.forName("ISO-8859-1"), charsetUtils.forName("iso-8859-1"));
        assertEquals(Charset.forName("CP1252"), charsetUtils.forName("cp1252"));
        assertEquals(Charset.forName("Windows-1252"), charsetUtils.forName("windows-1252"));
    }

    @Test
    public void testForNameWithISONamePattern() {
        assertEquals(Charset.forName("ISO-8859-1"), charsetUtils.forName("iso 8859-1"));
        assertEquals(Charset.forName("ISO-8859-2"), charsetUtils.forName("iso 8859-2"));
    }

    @Test
    public void testForNameWithCPNamePattern() {
        assertEquals(Charset.forName("CP1252"), charsetUtils.forName("cp1252"));
        assertEquals(Charset.forName("CP1251"), charsetUtils.forName("cp1251"));
    }

    @Test
    public void testForNameWithWINNamePattern() {
        assertEquals(Charset.forName("Windows-1252"), charsetUtils.forName("win1252"));
        assertEquals(Charset.forName("Windows-1251"), charsetUtils.forName("win1251"));
    }

    @Test
    public void testForNameWithICU() throws Exception {
        Method getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        getCharsetICU.setAccessible(true);
        Method isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        isSupportedICU.setAccessible(true);
        when(getCharsetICU.invoke(null, "utf-8")).thenReturn(Charset.forName("UTF-8"));
        when(isSupportedICU.invoke(null, "utf-8")).thenReturn(true);
        assertEquals(Charset.forName("UTF-8"), charsetUtils.forName("utf-8"));
    }
}
