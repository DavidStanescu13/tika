public class CharsetUtils_isSupported_1_3_Test {

    private static Method isSupportedMethod;

    private static Method getCharsetICU;

    private static Method isSupportedICU;

    @BeforeAll
    public static void setUp() throws NoSuchMethodException {
        isSupportedMethod = CharsetUtils.class.getDeclaredMethod("isSupported", String.class);
        isSupportedMethod.setAccessible(true);
        getCharsetICU = CharsetUtils.class.getDeclaredField("getCharsetICU").get(null);
        isSupportedICU = CharsetUtils.class.getDeclaredField("isSupportedICU").get(null);
    }

    @Test
    public void testIsSupportedWithSupportedCharset() throws Exception {
        when((Method) getCharsetICU.invoke(null, "UTF-8")).thenReturn(Charset.forName("UTF-8"));
        when((Method) isSupportedICU.invoke(null, "UTF-8")).thenReturn(true);
        boolean result = (boolean) isSupportedMethod.invoke(null, "UTF-8");
        assertTrue(result);
    }

    @Test
    public void testIsSupportedWithUnsupportedCharset() throws Exception {
        when((Method) getCharsetICU.invoke(null, "INVALID-CHARSET")).thenReturn(null);
        when((Method) isSupportedICU.invoke(null, "INVALID-CHARSET")).thenReturn(false);
        boolean result = (boolean) isSupportedMethod.invoke(null, "INVALID-CHARSET");
        assertFalse(result);
    }

    @Test
    public void testIsSupportedWithIllegalCharsetNameException() throws Exception {
        when((Method) getCharsetICU.invoke(null, "INVALID-CHARSET")).thenReturn(null);
        when((Method) isSupportedICU.invoke(null, "INVALID-CHARSET")).thenThrow(new IllegalCharsetNameException("INVALID-CHARSET"));
        boolean result = (boolean) isSupportedMethod.invoke(null, "INVALID-CHARSET");
        assertFalse(result);
    }

    @Test
    public void testIsSupportedWithIllegalArgumentException() throws Exception {
        when((Method) getCharsetICU.invoke(null, null)).thenReturn(null);
        when((Method) isSupportedICU.invoke(null, null)).thenReturn(false);
        boolean result = (boolean) isSupportedMethod.invoke(null, null);
        assertFalse(result);
    }

    @Test
    public void testIsSupportedWithUnexpectedException() throws Exception {
        when((Method) getCharsetICU.invoke(null, "INVALID-CHARSET")).thenReturn(null);
        when((Method) isSupportedICU.invoke(null, "INVALID-CHARSET")).thenThrow(new RuntimeException("Unexpected exception"));
        boolean result = (boolean) isSupportedMethod.invoke(null, "INVALID-CHARSET");
        assertFalse(result);
    }
}
