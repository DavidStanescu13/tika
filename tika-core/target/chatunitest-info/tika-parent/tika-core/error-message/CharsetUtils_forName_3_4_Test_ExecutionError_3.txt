public class CharsetUtils_forName_3_4_Test {

    private CharsetUtils charsetUtils;

    private Map<String, Charset> commonCharsets;

    private Method getCharsetICU;

    private Method isSupportedICU;

    @BeforeEach
    public void setUp() throws NoSuchMethodException {
        charsetUtils = new CharsetUtils();
        commonCharsets = new HashMap<>();
        commonCharsets.put("utf-8", Charset.forName("UTF-8"));
        commonCharsets.put("iso-8859-1", Charset.forName("ISO-8859-1"));
        commonCharsets.put("cp1252", Charset.forName("CP1252"));
        commonCharsets.put("windows-1252", Charset.forName("Windows-1252"));
        getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        getCharsetICU.setAccessible(true);
        isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        isSupportedICU.setAccessible(true);
    }

    @Test
    public void testForNameWithValidName() throws Exception {
        when(getCharsetICU.invoke(null, "UTF-8")).thenReturn(Charset.forName("UTF-8"));
        Charset charset = charsetUtils.forName("UTF-8");
        assertEquals(Charset.forName("UTF-8"), charset);
    }

    @Test
    public void testForNameWithInvalidName() throws Exception {
        when(getCharsetICU.invoke(null, "INVALID")).thenReturn(null);
        assertThrows(IllegalCharsetNameException.class, () -> charsetUtils.forName("INVALID"));
    }

    @Test
    public void testForNameWithCPXError() throws Exception {
        when(getCharsetICU.invoke(null, "cp-1252")).thenReturn(Charset.forName("CP1252"));
        Charset charset = charsetUtils.forName("cp-1252");
        assertEquals(Charset.forName("CP1252"), charset);
    }

    @Test
    public void testForNameWithICUUnsupported() throws Exception {
        when(getCharsetICU.invoke(null, "UTF-8")).thenReturn(null);
        when(isSupportedICU.invoke(null, "UTF-8")).thenReturn(false);
        assertThrows(IllegalCharsetNameException.class, () -> charsetUtils.forName("UTF-8"));
    }

    @Test
    public void testForNameWithISO8859XError() throws Exception {
        when(getCharsetICU.invoke(null, "iso-8859-1")).thenReturn(Charset.forName("ISO-8859-1"));
        Charset charset = charsetUtils.forName("iso-8859-1");
        assertEquals(Charset.forName("ISO-8859-1"), charset);
    }

    @Test
    public void testForNameWithWinXError() throws Exception {
        when(getCharsetICU.invoke(null, "windows-1252")).thenReturn(Charset.forName("Windows-1252"));
        Charset charset = charsetUtils.forName("windows-1252");
        assertEquals(Charset.forName("Windows-1252"), charset);
    }

    @Test
    public void testForNameWithNullName() throws Exception {
        assertThrows(IllegalArgumentException.class, () -> charsetUtils.forName(null));
    }

    @Test
    public void testForNameWithICUCharset() throws Exception {
        when(getCharsetICU.invoke(null, "UTF-8")).thenReturn(Charset.forName("UTF-8"));
        when(isSupportedICU.invoke(null, "UTF-8")).thenReturn(true);
        Charset charset = charsetUtils.forName("UTF-8");
        assertEquals(Charset.forName("UTF-8"), charset);
    }
}
