package org.apache.tika.utils;

import org.apache.tika.utils.FileProcessResult;
import org.apache.tika.utils.ProcessUtils;
import org.apache.tika.utils.StreamGobbler;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.ByteArrayInputStream;
import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class ProcessUtils_execute_5_1_Test {

    @Mock
    private StreamGobbler errGobbler;

    @Mock
    private Process process;

    @InjectMocks
    private ProcessUtils processUtils;

    private Path stdoutRedirect;

    private ProcessBuilder pb;

    private long timeoutMillis;

    private int maxStdErrBuffer;

    @BeforeEach
    public void setUp() throws IOException {
        stdoutRedirect = Files.createTempFile("stdout", ".txt");
        pb = new ProcessBuilder("echo", "Hello, World!");
        timeoutMillis = 1000;
        maxStdErrBuffer = 1024;
    }

    @Test
    public void testExecute_Success() throws IOException, InterruptedException {
        when(process.waitFor(timeoutMillis, TimeUnit.MILLISECONDS)).thenReturn(true);
        when(process.exitValue()).thenReturn(0);
        when(errGobbler.getStreamLength()).thenReturn(13L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(List.of("Hello, World!"));
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertEquals(13L, result.stderrLength);
        assertEquals(13L, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("Hello, World!\n", result.stderr);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecute_Timeout() throws IOException, InterruptedException {
        when(process.waitFor(timeoutMillis, TimeUnit.MILLISECONDS)).thenReturn(false);
        when(process.exitValue()).thenReturn(-1);
        when(errGobbler.getStreamLength()).thenReturn(0L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(List.of());
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertTrue(result.isTimeout);
        assertEquals(-1, result.exitValue);
        assertEquals(0L, result.stderrLength);
        assertEquals(0L, result.stdoutLength);
        assertEquals("", result.stderr);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecute_IOException() throws IOException {
        when(process.waitFor(timeoutMillis, TimeUnit.MILLISECONDS)).thenThrow(new IOException("Test IOException"));
        assertThrows(IOException.class, () -> processUtils.execute(pb, timeoutMillis, stdoutRedirect, maxStdErrBuffer));
    }

    @Test
    public void testExecute_InterruptedException() throws IOException, InterruptedException {
        when(process.waitFor(timeoutMillis, TimeUnit.MILLISECONDS)).thenThrow(new InterruptedException("Test InterruptedException"));
        FileProcessResult result = processUtils.execute(pb, timeoutMillis, stdoutRedirect, maxStdErrBuffer);
        assertEquals(-1000, result.exitValue);
        assertTrue(result.isTimeout);
    }
}
