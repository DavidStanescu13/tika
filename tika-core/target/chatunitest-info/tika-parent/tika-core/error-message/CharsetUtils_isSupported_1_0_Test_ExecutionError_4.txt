public class CharsetUtils_isSupported_1_0_Test {

    private CharsetUtils charsetUtils;

    private Method getCharsetICU;

    private Method isSupportedICU;

    @BeforeEach
    public void setUp() throws NoSuchMethodException {
        charsetUtils = new CharsetUtils();
        getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        getCharsetICU.setAccessible(true);
        isSupportedICU.setAccessible(true);
    }

    @Test
    public void testIsSupportedWithUnexpectedException() throws Exception {
        when(isSupportedICU.invoke(null, "UTF-8")).thenThrow(new Exception("Unexpected exception"));
        assertFalse(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithNoICUSupportAndUnsupportedCharset() throws Exception {
        when(isSupportedICU.invoke(null, "INVALID_CHARSET")).thenReturn(false);
        assertFalse(charsetUtils.isSupported("INVALID_CHARSET"));
    }

    @Test
    public void testIsSupportedWithSupportedCharset() throws Exception {
        when(isSupportedICU.invoke(null, "UTF-8")).thenReturn(true);
        assertTrue(charsetUtils.isSupported("UTF-8"));
    }

    @Test
    public void testIsSupportedWithIllegalArgumentException() throws Exception {
        when(isSupportedICU.invoke(null, null)).thenThrow(new IllegalArgumentException("Null charset name"));
        assertFalse(charsetUtils.isSupported(null));
    }

    @Test
    public void testIsSupportedWithIllegalCharsetNameException() throws Exception {
        when(isSupportedICU.invoke(null, "INVALID_CHARSET")).thenThrow(new IllegalCharsetNameException("Invalid charset name"));
        assertFalse(charsetUtils.isSupported("INVALID_CHARSET"));
    }

    @Test
    public void testIsSupportedWithUnsupportedCharset() throws Exception {
        when(isSupportedICU.invoke(null, "INVALID_CHARSET")).thenReturn(false);
        assertFalse(charsetUtils.isSupported("INVALID_CHARSET"));
    }

    @Test
    public void testIsSupportedWithNoICUSupport() throws Exception {
        when(isSupportedICU.invoke(null, "UTF-8")).thenReturn(false);
        assertTrue(charsetUtils.isSupported("UTF-8"));
    }
}
