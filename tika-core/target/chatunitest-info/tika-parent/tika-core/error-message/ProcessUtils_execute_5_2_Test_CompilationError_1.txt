import java.util.UUID;

import java.util.concurrent.TimeUnit;
import org.apache.tika.utils.FileProcessResult;
import org.apache.tika.utils.ProcessUtils;
import org.apache.tika.utils.StreamGobbler;
import org.apache.tika.utils.StringUtils;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.util.concurrent.TimeoutException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
public class ProcessUtils_execute_5_2_Test {

    @Mock
    private ProcessBuilder processBuilder;

    @Mock
    private Process process;

    @Mock
    private StreamGobbler errGobbler;

    @InjectMocks
    private ProcessUtils processUtils;

    private Path stdoutRedirect;

    private ConcurrentHashMap<String, Process> processMap;

    @BeforeEach
    public void setUp() throws IOException {
        stdoutRedirect = Files.createTempFile("test", ".txt");
        processMap = new ConcurrentHashMap<>();
        when(processBuilder.start()).thenReturn(process);
        when(process.getErrorStream()).thenReturn(new ByteArrayInputStream("error".getBytes()));
        when(process.getInputStream()).thenReturn(new ByteArrayInputStream("output".getBytes()));
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(true);
        when(process.exitValue()).thenReturn(0);
        when(process.isAlive()).thenReturn(false);
        when(errGobbler.getStreamLength()).thenReturn(5L);
        when(errGobbler.getIsTruncated()).thenReturn(false);
        when(errGobbler.getLines()).thenReturn(java.util.Arrays.asList("error"));
        processUtils.PROCESS_MAP = processMap;
    }

    @Test
    public void testExecute() throws IOException, InterruptedException {
        FileProcessResult result = processUtils.execute(processBuilder, 1000, stdoutRedirect, 1000);
        assertEquals(1000, result.processTimeMillis);
        assertEquals(5, result.stderrLength);
        assertEquals(5, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("error", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }

    @Test
    public void testExecuteWithTimeout() throws IOException, InterruptedException {
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenReturn(false);
        when(process.isAlive()).thenReturn(true);
        FileProcessResult result = processUtils.execute(processBuilder, 100, stdoutRedirect, 1000);
        assertTrue(result.isTimeout);
        assertEquals(-1, result.exitValue);
    }

    @Test
    public void testExecuteWithException() throws IOException, InterruptedException {
        when(process.waitFor(anyLong(), any(TimeUnit.class))).thenThrow(new InterruptedException());
        FileProcessResult result = processUtils.execute(processBuilder, 1000, stdoutRedirect, 1000);
        assertEquals(-1000, result.exitValue);
    }

    @Test
    public void testExecuteWithNonExistentParentDirectory() throws IOException, InterruptedException {
        Path nonExistentParent = Paths.get("/non/existent/parent");
        when(stdoutRedirect.getParent()).thenReturn(nonExistentParent);
        FileProcessResult result = processUtils.execute(processBuilder, 1000, stdoutRedirect, 1000);
        assertTrue(Files.exists(nonExistentParent));
        assertEquals(1000, result.processTimeMillis);
        assertEquals(5, result.stderrLength);
        assertEquals(5, result.stdoutLength);
        assertFalse(result.isTimeout);
        assertEquals(0, result.exitValue);
        assertEquals("error", result.stderr);
        assertFalse(result.stdoutTruncated);
        assertFalse(result.stderrTruncated);
    }
}
