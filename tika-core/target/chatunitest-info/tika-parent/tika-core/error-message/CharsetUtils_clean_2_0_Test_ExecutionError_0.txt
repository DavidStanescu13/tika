        commonCharsets.put("utf-8", Charset.forName("UTF-8"));
        commonCharsets.put("iso-8859-1", Charset.forName("ISO-8859-1"));

    @Test
    public void testCleanWithValidCharset() throws Exception {
        String charsetName = "UTF-8";
        String result = charsetUtils.clean(charsetName);
        assertEquals("UTF-8", result);
    }

    public void testCleanWithInvalidCharset() throws Exception {
        String charsetName = "INVALID";
        String result = charsetUtils.clean(charsetName);
        assertNull(result);
    }

    }

        String result = charsetUtils.clean(charsetName);
        assertEquals("UTF-8", result);
    }

    @Test

    @Test
    public void testCleanWithICU() throws Exception {
        String charsetName = "UTF-8";
        Method isSupportedICU = CharsetUtils.class.getDeclaredMethod("isSupportedICU");
        isSupportedICU.setAccessible(true);
        when(isSupportedICU.invoke(null)).thenReturn(true);
        Method getCharsetICU = CharsetUtils.class.getDeclaredMethod("getCharsetICU");
        getCharsetICU.setAccessible(true);
        when(getCharsetICU.invoke(null, charsetName)).thenReturn(Charset.forName("UTF-8"));
        String result = charsetUtils.clean(charsetName);
        assertEquals("UTF-8", result);
    }

        assertNull(result);
    }
