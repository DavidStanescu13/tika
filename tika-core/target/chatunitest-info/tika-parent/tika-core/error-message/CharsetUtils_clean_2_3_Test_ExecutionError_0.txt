        String result = CharsetUtils.clean(charsetName);
        assertEquals(charsetName, result);
    }

    @Test
    public void testCleanWithInvalidCharset() throws Exception {

        assertNull(result);
    }

        assertNull(result);
    }

        String result = CharsetUtils.clean(charsetName);
        assertEquals(charsetName.trim(), result);
    }
